from getpass import getpass
from io import BytesIO
from pathlib import Path
from time import monotonic, sleep

from evaluation.run_scenario import ApiClient

api = ApiClient('https://localhost:8080/api-gateway', verify_tls=False, timeout=120)

api.process(project)
base = f'/projects/{api.user_id}/{project.project_id}'
deadline = monotonic() + 180
while monotonic() < deadline:
    state = api._request('GET', base + '/processing-status').json()
    if state and state['state'] == 'completed':
        break
    if state and state['state'] in ('failed', 'cancelled'):
        raise SystemExit(state['state'])
    sleep(0.25)
else:
    raise SystemExit('Processing timed out')

print(f'Run token: {state["runToken"]}', flush=True)
sleep(3)
outputs = api._request('GET', base + '/process/url').json()['imgs']
assert len(outputs) == len({item['og_img_id'] for item in outputs}) == 2
assert len({item['result_id'] for item in outputs}) == 2
for item in outputs:
    response = api.session.get(item['url'], verify=api.verify_tls, timeout=api.timeout)
    response.raise_for_status()
    assert response.content and item['type'] == 'video'
with ZipFile(BytesIO(api._request('GET', base + '/process').content)) as archive:
    assert sorted(archive.namelist()) == sorted(item['name'] for item in outputs)
    assert archive.testzip() is None
